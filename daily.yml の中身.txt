name: 相場データ自動更新

# 平日の日本時間 朝6:30（UTC 21:30）に実行。
# 日本市場の前日分＋米国市場の当日分をまとめて取得する。
# 手動実行: Actionsタブ → このワークフロー → Run workflow
on:
  schedule:
    - cron: '30 21 * * 1-5'
  workflow_dispatch:

permissions:
  contents: write

jobs:
  update:
    runs-on: ubuntu-latest
    steps:
      - uses: actions/checkout@v4

      - uses: actions/setup-python@v5
        with:
          python-version: '3.12'

      - name: 依存ライブラリ
        run: pip install -r scripts/requirements.txt

      - name: データ取得（成功率80%未満なら中断し前日データを維持）
        run: python scripts/fetch_data.py

      - name: コミット＆プッシュ
        run: |
          git config user.name "data-bot"
          git config user.email "41898282+github-actions[bot]@users.noreply.github.com"
          git add data/
          git diff --cached --quiet || (git commit -m "data: $(date -u +%Y-%m-%d)" && git push)
